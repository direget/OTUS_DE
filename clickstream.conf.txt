input {
  file {
    path => "/app/weblog.csv"
    start_position => "beginning"
    sincedb_path => "/dev/null"
  }
}

filter {
  csv {
    columns => ["IP", "Time", "URL", "Status"]
    separator => ","
  }

  date {
    match => [ "Time", "[dd/MMM/yyyy:HH:mm:ss", "dd/MMM/yyyy:HH:mm:ss" ]
    locale => "en"
    target => "eventtime"
  }

  mutate {
    convert => { "Status" => "integer" }
    strip => ["IP", "Time", "URL"]
    remove_field => ["path", "message", "host", "@version"]
  }
}

output {
  stdout {
    codec => rubydebug
  }

  elasticsearch {
    hosts => ["http://elasticsearch:9200"]
    index  => "clickstream-csv"
  }
}
